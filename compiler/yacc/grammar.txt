letter = ([a-z] | [A-Z]);
digit = [0-9];
number = digit+;
character = (letter | digit);
identifier = (letter | "_") (character | "_")*;
program = { ["pub"] declaration };
statements = statements statement;
statement = compound
			| "if" condition compound
			| "loop" compound
			| match identifier {(expression ":" compound)}
			| "break"
			| "continue"
			| "defer" [expression]
			| "return" [expression]
			| declaration
			;
compound = "{" statements "}";
declaration = ["mut" | "con"] identifier identifier ["=" expression]
			  | identifier [identifier] "(" {args} ")" [compound]
			  | "enum" identifier "{" {(identifier [("(" identifier {"," identifier} ")")])} "}" 
			  | identifier [identifier] [(":" struct {struct})] "{" {(var | func)} "}"
			  | "type" identifier = expression
			  ;
expression = number
			 | "(" expression")"
			 | identifier "(" {(expression{"," expression})} ")"
			 | compound
			 | identifier "{" [(identifier ":" expression){","(identifier ":" expression)}] "}"
			 | identifier "(" [identifier{"," identifier}] ")"
			 | expression bop expression
			 | ((preuop expression) | (expression postuop))
			 | identifier{"|" identifier}
			 | identifier{"&" identifier}
			 | "<" {identifier [":" identifier{"," identifier}]} ">"
			 ;
bop = "+" | "-" | "*" | "/" | "%" | "|" | "&" | "&&" | "||";
preuop = "*" | "&" | "-" | "++" | "--";
postuop = "++" | "--";
