use "std/c"

pub type DynMode = i32
pub const DYN_MODE_OFF: DynMode = 0
pub const DYN_MODE_ON: DynMode = 4

pub const DYN_LIMIT: u64 = 64
pub const DYN_NAME: []const u8 = "dyn"

pub struct DynPoint {
  x: i32,
  y: i32,
}

pub struct DynPair {
  left: i32,
  right: i32,
}

pub struct DynInner {
  value: i32,
}

pub struct DynNested {
  inner: DynInner,
}

pub extern fn dyn_point_sum "dyn_point_sum"(point: *const DynPoint) i32
pub extern fn dyn_value_reset "dyn_value_reset"(value: rawptr)
pub extern fn dyn_sum "dyn_sum"(count: u32, ...) f64
pub extern fn dyn_visit "dyn_visit"(callback: *fn(i64, rawptr) i64, context: rawptr)

// Unsupported declarations (write explicit bindings or a C adapter):
// function dyn_value_get: parameter or return type is not representable
// macro DYN_SHIFTED: expression is not a standalone integer or string literal
// macro DYN_SQUARE: function-like macros require a C wrapper
// struct <anonymous>: anonymous record has no stable Dyn type name
// struct DynAnonymous: anonymous fields are not representable
// struct DynFlags: bitfields require an explicit ABI adapter
// union DynValue: by-value union layout is not representable
