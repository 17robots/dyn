<Statement> ::= <VariableDeclaration> | <VariableInitialization> | <EnumDeclaration> | <FunctionDeclaration> | <FunctionCall> 


// Variable Declaration
<VariableDeclaration> ::= <MutableVariableDeclaration> | <StandardVariableDeclaration>

<StandardVariableDeclaration> ::= <StandardVariableDeclareAndAssign> | <StandardVariableDeclare>

<StandardVariableDeclareAndAssign> ::= <Type> <Identifier> `=` <Value> `;`

<StandardVariableDeclare> ::= <Type> <Identifier> ';'

<MutableVariableDeclaration> ::= `mut` <StandardVariableDeclaration> 


// variable initialization
<VariableInitialization> ::= <StandardVariableInitialization> | <MutableVariableReassign>

<StandardVariableInitialization> ::= <StandardVariableDeclareAndAssign> | <StandardVariableInit>

<StandardVariableInit> ::= <Identifier> `=` <Value> `;`

<MutableVariableReassign> ::= <Identifier> `=` <Value> `;`


// function declaration
<FunctionDeclaration> ::= <Type> <Identifier> `(` <Parameter> { `,` <Parameter> }* `)` <FunctionBody>

<FunctionBody> ::= `{` {<Statement>}* `}`

<Parameter> ::= <Type> <Identifier>


// function call 
<FunctionCall> ::= <Identifier> `(` <Argument> { `,` <Argument> }* `);`

<Argument> ::= 


// enum declaration
<EnumDeclaration> ::= `enum` <Identifier> `{` <EnumMember> `}`

<EnumMember> ::= <StandardEnumMember> | <PartneredEnumMember>

<StandardEnumMember> ::= <Identifier>

<PartneredEnumMember> ::= <Identifier> `(` <Type> `)`

<Expression> ::= 


// type
<Type> ::= <PrimitiveType> | <CustomType>
